## 🗄️ Sample DB Design — ParkVision Schema

</div>

<br/>

```sql
-- ════════════════════════════════════════════════════════════════
--  PARKVISION — Database Schema
--  SQLite / PostgreSQL compatible
-- ════════════════════════════════════════════════════════════════

-- ── PARKING SPACES TABLE ─────────────────────────────────────────
CREATE TABLE parking_spaces (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    space_code  VARCHAR(10)  NOT NULL UNIQUE,   -- e.g. "A-01", "B-12"
    zone        VARCHAR(50)  NOT NULL,           -- "VIP", "General", "Reserved"
    is_occupied BOOLEAN      DEFAULT FALSE,
    created_at  TIMESTAMP    DEFAULT CURRENT_TIMESTAMP
);

-- ── VEHICLES TABLE ───────────────────────────────────────────────
CREATE TABLE vehicles (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    plate        VARCHAR(20) NOT NULL UNIQUE,    -- "ABC-123"
    owner_name   VARCHAR(100),
    vehicle_type VARCHAR(50),                    -- "car", "motorcycle", "truck"
    created_at   TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ── ENTRY / EXIT RECORDS ─────────────────────────────────────────
CREATE TABLE parking_records (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    vehicle_id      INTEGER NOT NULL REFERENCES vehicles(id),
    space_id        INTEGER NOT NULL REFERENCES parking_spaces(id),
    entry_time      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    exit_time       TIMESTAMP,
    duration_mins   INTEGER,                     -- calculated on exit
    plate_raw       VARCHAR(20),                 -- raw OCR output
    confidence      FLOAT,                       -- PlateRecognizer confidence score
    camera_id       VARCHAR(20),                 -- which camera detected it
    status          VARCHAR(20) DEFAULT 'active' -- 'active', 'completed', 'error'
);

-- ── PLATE RECOGNITION LOGS ───────────────────────────────────────
CREATE TABLE recognition_logs (
    id            INTEGER PRIMARY KEY AUTOINCREMENT,
    timestamp     TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    raw_plate     VARCHAR(30),
    cleaned_plate VARCHAR(20),
    confidence    FLOAT,
    image_path    VARCHAR(255),
    matched       BOOLEAN DEFAULT FALSE,
    record_id     INTEGER REFERENCES parking_records(id)
);

-- ── USEFUL VIEWS ─────────────────────────────────────────────────
CREATE VIEW active_vehicles AS
    SELECT
        v.plate,
        v.vehicle_type,
        ps.space_code,
        ps.zone,
        pr.entry_time,
        ROUND((JULIANDAY('now') - JULIANDAY(pr.entry_time)) * 24 * 60) AS minutes_parked
    FROM parking_records pr
    JOIN vehicles v         ON pr.vehicle_id = v.id
    JOIN parking_spaces ps  ON pr.space_id   = ps.id
    WHERE pr.status = 'active'
    ORDER BY pr.entry_time DESC;

CREATE VIEW daily_summary AS
    SELECT
        DATE(entry_time)       AS day,
        COUNT(*)               AS total_entries,
        COUNT(exit_time)       AS completed,
        AVG(duration_mins)     AS avg_duration_mins,
        MAX(confidence)        AS best_confidence
    FROM parking_records
    GROUP BY DATE(entry_time)
    ORDER BY day DESC;
```

```python
from fastapi            import FastAPI, HTTPException, Depends
from fastapi.middleware.cors import CORSMiddleware
from pydantic           import BaseModel, validator
from typing             import List, Optional
import numpy            as np
import joblib
import logging
from datetime           import datetime

# ── App Setup ─────────────────────────────────────────────────────
app = FastAPI(
    title       = "Vexor Prediction API",
    description = "Binary ML prediction platform by frannnnuser",
    version     = "2.0.0",
    docs_url    = "/docs",
)

app.add_middleware(CORSMiddleware, allow_origins=["*"],
                   allow_methods=["*"], allow_headers=["*"])

logger = logging.getLogger("vexor")

# ── Load Model ─────────────────────────────────────────────────────
model    = joblib.load("models/binary_classifier.pkl")
pipeline = joblib.load("models/preprocessing_pipeline.pkl")

# ── Request / Response Schemas ─────────────────────────────────────
class PredictRequest(BaseModel):
    features: List[float]
    model_version: Optional[str] = "latest"

    @validator("features")
    def validate_features(cls, v):
        if len(v) == 0:
            raise ValueError("Features list cannot be empty")
        if len(v) > 100:
            raise ValueError("Too many features (max 100)")
        return v

class PredictResponse(BaseModel):
    prediction:  int           # 0 or 1
    confidence:  float         # 0.0 → 1.0
    label:       str           # "positive" or "negative"
    timestamp:   str
    model_ver:   str

# ── Prediction Endpoint ────────────────────────────────────────────
@app.post("/predict", response_model=PredictResponse, tags=["Prediction"])
async def predict(request: PredictRequest):
    """
    Submit feature vector → get binary prediction.
    Returns class (0/1), confidence score, and human-readable label.
    """
    try:
        # Preprocess
        features_array = np.array(request.features).reshape(1, -1)
        processed      = pipeline.transform(features_array)

        # Predict
        prediction     = int(model.predict(processed)[0])
        proba          = model.predict_proba(processed)[0]
        confidence     = float(max(proba))

        logger.info(f"Prediction: {prediction} | Confidence: {confidence:.3f}")

        return PredictResponse(
            prediction = prediction,
            confidence = round(confidence, 4),
            label      = "positive" if prediction == 1 else "negative",
            timestamp  = datetime.utcnow().isoformat() + "Z",
            model_ver  = request.model_version,
        )

    except Exception as e:
        logger.error(f"Prediction failed: {str(e)}")
        raise HTTPException(status_code=500, detail=str(e))

# ── Health Check ───────────────────────────────────────────────────
@app.get("/health", tags=["System"])
async def health():
    return {
        "status"   : "ok",
        "model"    : "loaded",
        "timestamp": datetime.utcnow().isoformat() + "Z",
        "version"  : "2.0.0",
    }
```

## ⚙️ GitHub Actions — CI/CD Pipeline

</div>

<br/>

```yaml
# ══════════════════════════════════════════════════════════════════
#  .github/workflows/ci.yml
#  Continuous Integration for Vexor ML API
#  Runs on every push to main + all pull requests
# ══════════════════════════════════════════════════════════════════

name: Vexor CI Pipeline

on:
  push:
    branches: [ main, develop ]
  pull_request:
    branches: [ main ]

env:
  PYTHON_VERSION: "3.11"

jobs:

  # ── Job 1: Code Quality ─────────────────────────────────────
  lint:
    name: Lint & Format Check
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: ${{ env.PYTHON_VERSION }}
          cache: pip

      - name: Install linting tools
        run: pip install flake8 black isort

      - name: Run flake8
        run: flake8 app/ --max-line-length=100 --ignore=E501

      - name: Check black formatting
        run: black --check app/

      - name: Check import ordering
        run: isort --check-only app/

  # ── Job 2: Tests ────────────────────────────────────────────
  test:
    name: Run Tests
    runs-on: ubuntu-latest
    needs: lint
    steps:
      - uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: ${{ env.PYTHON_VERSION }}
          cache: pip

      - name: Install dependencies
        run: pip install -r requirements.txt -r requirements-dev.txt

      - name: Run pytest
        run: |
          pytest tests/ -v \
            --cov=app \
            --cov-report=term-missing \
            --cov-fail-under=70
        env:
```

```python
# ══════════════════════════════════════════════════════════════════
#  parkvision/core/plate_detector.py
#  Real-time license plate detection & recognition
#  Uses OpenCV + PlateRecognizer API
# ══════════════════════════════════════════════════════════════════

import cv2
import requests
import base64
import logging
from dataclasses import dataclass
from typing      import Optional, List
from datetime    import datetime

logger = logging.getLogger("parkvision.detector")


@dataclass
class PlateResult:
    plate:       str
    confidence:  float
    bounding_box: dict
    detected_at:  str


class PlateDetector:
    """
    Real-time license plate detector.
    Combines OpenCV preprocessing with PlateRecognizer API for OCR.
    """

    def __init__(self, api_token: str, camera_index: int = 0):
        self.api_token    = api_token
        self.camera_index = camera_index
        self.api_url      = "https://api.platerecognizer.com/v1/plate-reader/"
        self.cap          = None

    # ── Camera Control ──────────────────────────────────────────
    def start(self) -> None:
        self.cap = cv2.VideoCapture(self.camera_index)
        self.cap.set(cv2.CAP_PROP_FRAME_WIDTH,  1280)
        self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 720)
        logger.info(f"Camera {self.camera_index} initialized")

    def stop(self) -> None:
        if self.cap:
            self.cap.release()
        cv2.destroyAllWindows()
        logger.info("Camera released")

    # ── Frame Preprocessing ─────────────────────────────────────
    def preprocess_frame(self, frame) -> bytes:
        """Convert OpenCV frame to JPEG bytes for API submission."""
        # Denoise + sharpen for better OCR
        denoised  = cv2.fastNlMeansDenoisingColored(frame, None, 10, 10, 7, 21)
        kernel    = cv2.getStructuringElement(cv2.MORPH_RECT, (3, 3))
        sharpened = cv2.filter2D(denoised, -1, kernel)

        _, buffer = cv2.imencode(
            ".jpg", sharpened,
            [cv2.IMWRITE_JPEG_QUALITY, 90]
        )
        return buffer.tobytes()

    # ── PlateRecognizer API Call ─────────────────────────────────
    def recognize_plate(self, frame_bytes: bytes) -> List[PlateResult]:
        """Submit frame to PlateRecognizer API and parse results."""
        try:
            response = requests.post(
                self.api_url,
                files   = {"upload": ("frame.jpg", frame_bytes, "image/jpeg")},
                headers = {"Authorization": f"Token {self.api_token}"},
                timeout = 5,
            )
            response.raise_for_status()
            data    = response.json()
            results = []

            for result in data.get("results", []):
                plate = result.get("plate", "").upper().strip()
                score = result.get("score", 0.0)
                box   = result.get("box", {})

                if score >= 0.75:  # Only accept high-confidence results
                    results.append(PlateResult(
                        plate        = plate,
                        confidence   = round(score, 4),
                        bounding_box = box,
                        detected_at  = datetime.utcnow().isoformat() + "Z",
                    ))
                    logger.info(f"Plate detected: {plate} ({score:.2%})")

            return results

        except requests.exceptions.Timeout:
            logger.warning("PlateRecognizer API timeout")
            return []
        except Exception as e:
            logger.error(f"Recognition error: {e}")
            return []

    # ── Main Detection Loop ──────────────────────────────────────
    def run(self, callback=None) -> None:
        """
        Main loop: capture → preprocess → recognize → callback.
        Pass a callback function to handle each PlateResult.
        """
        self.start()
        frame_count  = 0
        process_every = 15  # Process every 15th frame for performance

        while True:
            ret, frame = self.cap.read()
            if not ret:
                logger.error("Failed to read frame — reconnecting...")
                break

            frame_count += 1

            if frame_count % process_every == 0:
                frame_bytes = self.preprocess_frame(frame)
                results     = self.recognize_plate(frame_bytes)

                for result in results:
                    if callback:
                        callback(result)

                    # Draw bounding box on frame
                    box = result.bounding_box
                    cv2.rectangle(
                        frame,
                        (box["xmin"], box["ymin"]),
                        (box["xmax"], box["ymax"]),
                        (0, 255, 0), 2
                    )
                    cv2.putText(
                        frame,
                        f"{result.plate}  {result.confidence:.0%}",
                        (box["xmin"], box["ymin"] - 10),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2
                    )

            cv2.imshow("ParkVision — Live Feed", frame)
            if cv2.waitKey(1) & 0xFF == ord("q"):
                break

        self.stop()
```

```javascript
    },
  },

// ── Rate Limiter ────────────────────────────────────────────────
const loginLimiter = rateLimit({
  windowMs : 15 * 60 * 1000,  // 15 minutes
  max      : 5,                // 5 attempts per window
  message  : {
    error  : "Too many login attempts. Please try again in 15 minutes.",
    code   : "RATE_LIMIT_EXCEEDED",
  },
  standardHeaders : true,
  legacyHeaders   : false,
  skipSuccessfulRequests: true,
});

// ── Session Auth Check ─────────────────────────────────────────
const requireAuth = (req, res, next) => {
  if (!req.session?.userId) {
    req.session.returnTo = req.originalUrl;
    return res.redirect("/auth/login");
  }
  next();
};

// ── Role-Based Access Control ──────────────────────────────────
const requireRole = (...roles) => (req, res, next) => {
  if (!req.session?.userId) {
    return res.status(401).json({ error: "Unauthorized" });
  }
  if (!roles.includes(req.session.userRole)) {
    return res.status(403).json({ error: "Forbidden — insufficient permissions" });
  }
  next();
};

// ── Session Fixation Prevention ────────────────────────────────
const regenerateSession = (req) =>
  new Promise((resolve, reject) =>
    req.session.regenerate((err) => (err ? reject(err) : resolve()))
  );

// ── Login Handler ──────────────────────────────────────────────
const handleLogin = async (req, res) => {
  const { username, password } = req.body;

  try {
    // 1. Validate inputs
    if (!username?.trim() || !password?.trim()) {
      return res.status(400).render("auth/login", {
        error: "Username and password are required",
      });
    }

    // 2. Fetch user from database
    const user = await db.query(
      "SELECT id, username, password_hash, role FROM users WHERE username = $1",
      [username.trim().toLowerCase()]
    );

    if (!user.rows.length) {
      // Timing-safe: always compare even if user doesn't exist
      await bcrypt.compare(password, "$2b$12$fakehashtopreventtiming");
      return res.status(401).render("auth/login", { error: "Invalid credentials" });
    }

    const dbUser = user.rows[0];

    // 3. Verify password
    const valid = await bcrypt.compare(password, dbUser.password_hash);
    if (!valid) {
      return res.status(401).render("auth/login", { error: "Invalid credentials" });
    }

    // 4. Regenerate session (prevents session fixation)
    await regenerateSession(req);

    // 5. Store session data
    req.session.userId    = dbUser.id;
    req.session.username  = dbUser.username;
    req.session.userRole  = dbUser.role;
    req.session.loginTime = new Date().toISOString();

    // 6. Redirect
    const returnTo = req.session.returnTo || "/dashboard";
    delete req.session.returnTo;
    res.redirect(returnTo);

  } catch (err) {
    console.error("[Auth] Login error:", err);
    res.status(500).render("auth/login", { error: "Server error — try again" });
  }
};
```
